-- Opaque text glyphs look brighter than tinted icon artwork at the same RGB value.
-- Blend terminology text tints back toward the charcoal root so they visually match the icons,
-- including when the user changes the active accent color.
local function terminologyTint(iconTint, strength)
    local base = Theme.Root
    return Color3.new(
        base.R + (iconTint.R - base.R) * strength,
        base.G + (iconTint.G - base.G) * strength,
        base.B + (iconTint.B - base.B) * strength
    )
end

local function addCorner(instance, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius)
    corner.Parent = instance
    return corner
end

local function addStroke(instance, color, transparency, thickness)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color or Theme.Stroke
    stroke.Transparency = transparency or 0
    stroke.Thickness = thickness or 1
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    stroke.Parent = instance
    return stroke
end

local function makeFrame(name, parent, size, position, color, transparency)
    local frame = Instance.new("Frame")
    frame.Name = name
    frame.Size = size
    frame.Position = position
    frame.BackgroundColor3 = color or Theme.Surface
    frame.BackgroundTransparency = transparency or 0
    frame.BorderSizePixel = 0
    frame.Parent = parent
    return frame
end

local function makeText(name, parent, text, size, position, fontSize, color, font, alignment)
    local label = Instance.new("TextLabel")
    label.Name = name
    label.BackgroundTransparency = 1
    label.BorderSizePixel = 0
    label.Text = text
    label.Size = size
    label.Position = position
    label.FontFace = font or FONT_REGULAR
    label.TextScaled = false
    label.RichText = false
    label.AutoLocalize = false
    label.TextSize = fontSize or 14
    label.TextColor3 = color or Theme.Text
    label.TextXAlignment = alignment or Enum.TextXAlignment.Left
    label.TextYAlignment = Enum.TextYAlignment.Center
    label.TextTruncate = Enum.TextTruncate.AtEnd
    label.Parent = parent
    return label
end

local function tween(instance, duration, properties)
    local animation = TweenService:Create(
        instance,
        TweenInfo.new(duration, Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
        properties
    )
    animation:Play()
    return animation
end

-- Responsive viewport fit. Desktop and landscape keep the original canvas; portrait
-- touch devices use a taller single-column layout sized to the available phone viewport.
local function updateScale()
    local camera = Workspace.CurrentCamera
    if not camera then return end
    local viewport = camera.ViewportSize

    local topInset, bottomInset = 0, 0
    local okInset, topLeft, bottomRight = pcall(function()
        return GuiService:GetGuiInset()
    end)
    if okInset then
        if typeof(topLeft) == "Vector2" then topInset = math.max(0, topLeft.Y) end
        if typeof(bottomRight) == "Vector2" then bottomInset = math.max(0, bottomRight.Y) end
    end

    -- Desktop and landscape retain the original 940x650 canvas. Portrait touch
    -- devices use a tall 420px-wide canvas so the interface can fill the phone instead
    -- of shrinking the complete landscape window to a tiny strip.
    local touchDevice = touchInputEnabled
    local portraitTouch = touchDevice and viewport.Y > viewport.X
    mobilePortraitLayout = portraitTouch

    local horizontalMargin = portraitTouch and 10 or (touchDevice and 14 or 28)
    local verticalMargin = portraitTouch and 10 or (touchDevice and 14 or 18)
    local safeTop = touchDevice and topInset or 0
    local safeBottom = touchDevice and bottomInset or 0
    local availableWidth = math.max(120, viewport.X - horizontalMargin * 2)
    local availableHeight = math.max(120, viewport.Y - safeTop - safeBottom - verticalMargin * 2)

    local canvasWidth, canvasHeight
    if portraitTouch then
        canvasWidth = 420
        -- Match the available phone aspect ratio. This makes the shell fill the safe
        -- viewport while preserving a stable 420px layout coordinate system.
        canvasHeight = math.max(720, math.floor(canvasWidth * availableHeight / availableWidth + 0.5))
        mobileCanvasHeight = canvasHeight
    else
        canvasWidth, canvasHeight = 940, 650
    end

    root.Size = UDim2.fromOffset(canvasWidth, canvasHeight)
    baseWindowSize = root.Size

    local fittedScale = math.min(availableWidth / canvasWidth, availableHeight / canvasHeight)
    local maximumScale = touchDevice and 1.5 or 1
    responsiveScaleValue = math.max(0.1, math.min(fittedScale, maximumScale)) * windowSizeMultiplier

    mobileSafeOffsetY = touchDevice and math.floor((safeTop - safeBottom) / 2) or 0
    if not windowHasCustomPosition then
        root.Position = UDim2.new(0.5, 0, 0.5, mobileSafeOffsetY)
    end

    uiScale.Scale = responsiveScaleValue * windowMotionFactor.Value

    -- Preserve the existing minimum touch target for switches when the shell must
    -- scale below 1.0. When the shell scales up, switches scale with the rest of UI.
    local switchCompensationScale = responsiveScaleValue < 1
        and (1 / math.max(responsiveScaleValue, 0.25)) or 1
    for _, switchTrack in ipairs(gui:GetDescendants()) do
        if switchTrack:IsA("GuiButton") and switchTrack.Name == "Toggle"
            and switchTrack:GetAttribute("SyllinseBaseSwitchWidth") then
            local switchScale = switchTrack:FindFirstChild("SyllinseSwitchScaleCompensation")
            if switchScale and switchScale:IsA("UIScale") then
                switchScale.Scale = switchCompensationScale
            end

            local switchRow = switchTrack.Parent
            while switchRow and switchRow ~= root do
                if switchRow.Name == "LabelFrame" then
                    local baseHeight = switchRow:GetAttribute("SyllinseToggleBaseRowHeight")
                    if type(baseHeight) == "number" and switchRow.Size.Y.Scale == 0 then
                        local minimumPhysicalHeight = touchDevice and 48 or 24
                        local targetHeight = math.max(baseHeight, minimumPhysicalHeight / math.max(responsiveScaleValue, 0.25))
                        switchRow.Size = UDim2.new(switchRow.Size.X.Scale, switchRow.Size.X.Offset, 0, math.ceil(targetHeight))
                        switchRow.ClipsDescendants = false
                    end
                    break
                end
                switchRow = switchRow.Parent
            end
        end
    end

    -- Keep a manually dragged shell within the newly fitted viewport after orientation
    -- changes or resolution changes, preserving the dedicated top-right drag handle.
    if windowHasCustomPosition then
        task.defer(function()
            if not root.Parent then return end
            local activeCamera = Workspace.CurrentCamera
            local viewportNow = activeCamera and activeCamera.ViewportSize or gui.AbsoluteSize
            local currentSize = root.AbsoluteSize
            local margin = 8
            local minX = currentSize.X / 2 + margin
            local maxX = viewportNow.X - currentSize.X / 2 - margin
            local minY = currentSize.Y / 2 + margin + (touchDevice and topInset or 0)
            local maxY = viewportNow.Y - currentSize.Y / 2 - margin - (touchDevice and bottomInset or 0)
            local cx = viewportNow.X * root.Position.X.Scale + root.Position.X.Offset
            local cy = viewportNow.Y * root.Position.Y.Scale + root.Position.Y.Offset
            if maxX < minX then cx = viewportNow.X / 2 else cx = math.clamp(cx, minX, maxX) end
            if maxY < minY then cy = (minY + maxY) / 2 else cy = math.clamp(cy, minY, maxY) end
            root.Position = UDim2.new(
                root.Position.X.Scale, cx - viewportNow.X * root.Position.X.Scale,
                root.Position.Y.Scale, cy - viewportNow.Y * root.Position.Y.Scale
            )
        end)
    end

    if layoutResponsiveShell then task.defer(layoutResponsiveShell) end
end

local function makeVectorFallback(name, parent, size, position, anchorPoint, zIndex, kind)
    size = size or 24
    local rootFallback = Instance.new("Frame")
    rootFallback.Name = name or "ImageFallback"
    rootFallback.Size = UDim2.fromOffset(size, size)
    rootFallback.Position = position or UDim2.fromOffset(0, 0)
    rootFallback.AnchorPoint = anchorPoint or Vector2.new(0, 0)
    rootFallback.BackgroundTransparency = 1
    rootFallback.BorderSizePixel = 0
    rootFallback.ZIndex = zIndex or 2
    rootFallback.Parent = parent

    local color = Theme.Muted
    local function line(x1, y1, x2, y2, thickness)
        local scale = size / 24
        local dx, dy = x2 - x1, y2 - y1
        local piece = Instance.new("Frame")
        piece.Name = "Stroke"
        piece.AnchorPoint = Vector2.new(0.5, 0.5)
        piece.Position = UDim2.fromOffset(((x1 + x2) / 2) * scale, ((y1 + y2) / 2) * scale)
        piece.Size = UDim2.fromOffset(math.max(1, math.sqrt(dx * dx + dy * dy) * scale), math.max(1, thickness * scale))
        piece.Rotation = math.deg(math.atan2(dy, dx))
        piece.BackgroundColor3 = color
        piece.BorderSizePixel = 0
        piece.ZIndex = rootFallback.ZIndex + 1
        piece.Parent = rootFallback
        addCorner(piece, math.max(1, math.floor(thickness * scale)))
        return piece

        local scale = size / 24

        c.BorderSizePixel = 0

local function recolorFallback(fallback, active, activeColor, idleColor)
    if not fallback then return end
    local color = active and (activeColor or Theme.IconActive) or (idleColor or Theme.IconIdle)
    for _, descendant in ipairs(fallback:GetDescendants()) do
        if descendant:IsA("Frame") then
            descendant.BackgroundColor3 = color
            descendant.BackgroundTransparency = active and 0 or 0.04
        elseif descendant:IsA("TextLabel") then
            descendant.TextColor3 = color
            descendant.TextTransparency = 0
        elseif descendant:IsA("UIStroke") then
            descendant.Color = color
            descendant.Transparency = active and 0 or 0.04
        end
    end
end

local function setRobloxImage(imageLabel, assetUri, _thumbSize, fallback)
    if not imageLabel or not assetUri or assetUri == "" then
        return
    end
    if not fallback and imageLabel.Parent then
        fallback = imageLabel.Parent:FindFirstChild("ImageFallback")
    end

    local source = tostring(assetUri)
    local assetId = source:match("(%d+)")
    local contentUri
    if source:match("^%s*rbxassetid://") then
        contentUri = source:gsub("^%s+", "")
    elseif source:match("^%s*%d+%s*$") then
        contentUri = "rbxassetid://" .. source:match("%d+")
    else
        contentUri = source
    end

    if assetId then imageLabel:SetAttribute("SyllinseAssetId", assetId) end
    imageLabel.ImageColor3 = Color3.new(1, 1, 1)
    imageLabel.ImageTransparency = 0
    imageLabel.ScaleType = Enum.ScaleType.Fit
    imageLabel.Visible = false
    if fallback then fallback.Visible = false end

    task.spawn(function()
        local function tryLoad(uri)
            local success = false
            local statusText = "no fetch callback status"
            imageLabel.Visible = false
            imageLabel.Image = uri
            local ok, err = pcall(function()
                ContentProvider:PreloadAsync({ imageLabel }, function(_contentId, fetchStatus)
                    statusText = tostring(fetchStatus)
                    if fetchStatus == Enum.AssetFetchStatus.Success then
                        success = true
                    end
                end)
            end)
            if not ok then
                return false, statusText, err
            end
            return success, statusText, nil
        end

        local loaded, status, err = tryLoad(contentUri)
        local usedThumbnail = false

        -- Only try a thumbnail if the direct image content failed. Keep the image hidden
        -- while loading so neither a broken-image placeholder nor a temporary generic icon flashes.
        if not loaded and assetId then
            imageLabel.Image = ""
            task.wait(0.1)
            local thumbUri = "rbxthumb://type=Asset&id=" .. assetId .. "&w=420&h=420"
            local thumbLoaded, thumbStatus, thumbErr = tryLoad(thumbUri)
            if thumbLoaded then
                loaded = true
                usedThumbnail = true
            else
                status = thumbStatus or status
                err = thumbErr or err
            end
        end

        if not imageLabel.Parent then return end
        if loaded then
            imageLabel.Visible = true
            if fallback then fallback.Visible = false end
            if usedThumbnail then
                imageLabel:SetAttribute("SyllinseLoadedAsThumbnail", true)
            end
            return
        end

        imageLabel.Image = ""
        imageLabel.Visible = false
        imageLabel:SetAttribute("SyllinseImageLoadFailed", true)
        if fallback then
            -- A fallback can fail after the current tab was selected; style it from the
            -- current selection immediately instead of waiting for the next tab click.
            local parentName = imageLabel.Parent and imageLabel.Parent.Name or ""
            local selected = parentName == (activeName .. "Button")
                or parentName == ("HeaderIconHolder_" .. activeName)
            local isLogo = assetId == LOGO_IMAGE_ID:match("(%d+)")
            if not isLogo then
                recolorFallback(fallback, selected)
            end
            fallback.Visible = true
        end
        local warningId = assetId or contentUri
        if not warnedAssetIds[warningId] then
            warnedAssetIds[warningId] = true
            if err then
                warn("[Syllinse] Could not preload image " .. tostring(warningId) .. ": " .. tostring(err) .. ". Keeping the vector fallback visible.")
            else
                warn("[Syllinse] Roblox could not fetch image asset " .. tostring(warningId) ..
                    " (" .. tostring(status) .. "). Keeping the vector fallback visible. Check the content ID and experience permissions.")
            end
        end
    end)
end

-- Create an image icon plus a drawn fallback. Broken Roblox image placeholders are never shown.
local function makeImageIcon(name, parent, assetUri, size, position, anchorPoint, zIndex, _color, fallbackSize, imageWidth, imageHeight)
    local id = tostring(assetUri or ""):match("(%d+)")
    local kind = ICON_KIND_BY_ID[id] or name or "Generic"
    local fallback = makeVectorFallback(
        "ImageFallback", parent, fallbackSize or size or 24, position or UDim2.fromOffset(0, 0),
        anchorPoint or Vector2.new(0, 0), zIndex or 2, kind
    )

    local image = Instance.new("ImageLabel")
    image.Name = name or "Icon"
    image.Size = UDim2.fromOffset(imageWidth or size or 24, imageHeight or size or 24)
    image.Position = position or UDim2.fromOffset(0, 0)
    image.AnchorPoint = anchorPoint or Vector2.new(0, 0)
    image.BackgroundTransparency = 1
    image.BorderSizePixel = 0
    image.Image = ""
    image.ImageColor3 = Color3.new(1, 1, 1)
    image.ImageTransparency = 0
    image.ScaleType = Enum.ScaleType.Fit
    image.ZIndex = (zIndex or 2) + 1
    image.Visible = false
    image.Parent = parent
    setRobloxImage(image, assetUri, nil, fallback)
    image:SetAttribute("SyllinseFallbackName", fallback.Name)
    return image
end

-- If the execution environment supports local custom assets and the PNG is in its working
-- directory, prefer the supplied original logo PNG. Otherwise use the Roblox asset ID.
-- This is optional and never blocks the UI from opening.
local function findLocalLogoAsset()
    local customAsset = (type(getcustomasset) == "function" and getcustomasset)
        or (type(getsynasset) == "function" and getsynasset)
    if type(customAsset) ~= "function" or type(isfile) ~= "function" then
        return nil
    end
    local paths = { "Syllinse_Logo.png", "Syllinse/Syllinse_Logo.png", "assets/Syllinse_Logo.png" }
    for _, path in ipairs(paths) do
        local okFile, exists = pcall(isfile, path)
        if okFile and exists then
            local okAsset, content = pcall(customAsset, path)
            if okAsset and type(content) == "string" and content ~= "" then
                return content
            end
        end
    end
    return nil
end

-- A small neutral vector gear for Settings because no Settings asset ID was supplied.
-- It is drawn from GUI primitives rather than a font glyph, avoiding missing-box icons.
local function makeGearIcon(parent, size, position, color, zIndex, anchorPoint)
    local container = Instance.new("Frame")
    container.Name = "GearIcon"
    container.Size = UDim2.fromOffset(size, size)
    container.Position = position
    container.AnchorPoint = anchorPoint or Vector2.new(0, 0)
    container.BackgroundTransparency = 1
    container.BorderSizePixel = 0
    container.ZIndex = zIndex or 3
    container.Parent = parent

    local parts = {}
    local center = size / 2
    local radius = size * 0.33
    local toothW = math.max(3, math.floor(size * 0.19))
    local toothH = math.max(4, math.floor(size * 0.28))
    for i = 0, 7 do
        local a = (math.pi * 2 * i / 8) - math.pi / 2
        local tooth = Instance.new("Frame")
        tooth.Name = "Tooth" .. tostring(i + 1)
        tooth.AnchorPoint = Vector2.new(0.5, 0.5)
        tooth.Size = UDim2.fromOffset(toothW, toothH)
        tooth.Position = UDim2.fromOffset(center + math.cos(a) * radius, center + math.sin(a) * radius)
        tooth.Rotation = i * 45
        tooth.BackgroundColor3 = color or Theme.Muted
        tooth.BorderSizePixel = 0
        tooth.ZIndex = container.ZIndex
        tooth.Parent = container
        addCorner(tooth, 1)
        table.insert(parts, tooth)
    end

    local ring = Instance.new("Frame")
    ring.Name = "GearRing"
    ring.AnchorPoint = Vector2.new(0.5, 0.5)
    ring.Size = UDim2.fromOffset(size * 0.56, size * 0.56)
    ring.Position = UDim2.fromScale(0.5, 0.5)
    ring.BackgroundTransparency = 1
    ring.BorderSizePixel = 0
    ring.ZIndex = container.ZIndex + 1
    ring.Parent = container
    addCorner(ring, size)
    local ringStroke = addStroke(ring, color or Theme.Muted, 0, 1.5)

    local centerDot = Instance.new("Frame")
    centerDot.Name = "GearCenter"
    centerDot.AnchorPoint = Vector2.new(0.5, 0.5)
    centerDot.Size = UDim2.fromOffset(math.max(3, size * 0.18), math.max(3, size * 0.18))
    centerDot.Position = UDim2.fromScale(0.5, 0.5)
    centerDot.BackgroundColor3 = color or Theme.Muted
    centerDot.BorderSizePixel = 0
    centerDot.ZIndex = container.ZIndex + 2
    centerDot.Parent = container
    addCorner(centerDot, size)

    return { Container = container, Parts = parts, RingStroke = ringStroke, Center = centerDot }
end

local function setActiveTab(name)
    activeName = name
    local found
    for _, item in ipairs(navItems) do
        if item.Name == name then
            found = item
        end
    end

    if headerTitle then
        headerTitle.Text = name
        -- Keep a small, consistent gap from the visible icon; set per category in navItems.
        headerTitle.Position = UDim2.fromOffset(found and found.HeaderTitleX or 51, 19)
    end
    updateHeaderIcon(name)
    if headerGear and found then
        headerGear.Container.Visible = found.Gear == true
        setGearIconColor(headerGear, Theme.Text)
    end

    for _, nav in ipairs(navButtons) do
        local selected = nav.Name == name
        local terminologyLabel = terminologyLabels[nav.Name]
        if terminologyLabel then
            -- Match the tab's current hover color while keeping labels independent.
            local hoveringThisTab = nav.Hovering == true
            tween(terminologyLabel, 0.14, {
                -- Text uses a muted perceived-match tint because opaque glyphs look
                -- brighter than translucent icon artwork at the same RGB value.
                TextColor3 = selected and terminologyTint(Theme.IconActive, 0.76)
                    or (hoveringThisTab and terminologyTint(Theme.IconHover, 0.86) or Theme.IconIdle),
            })
        end
        -- The mockup has no selected tile or accent bar; only the active icon brightens.
        tween(nav.Button, 0.14, {
            BackgroundColor3 = Theme.Root,
            BackgroundTransparency = 1,
        })
        if nav.Icon then
            -- Selected stays bright; a hovered, unselected icon uses the same muted
            -- hover tint as its separate terminology label.
            tween(nav.Icon, 0.14, {
                ImageColor3 = selected and Theme.IconActive
                    or (nav.Hovering and Theme.IconHover or Theme.IconIdle),
                ImageTransparency = 0,
            })
            setFallbackEmphasis(nav.Button, selected)
        end
        if nav.Gear then
            setGearIconColor(nav.Gear, selected and Theme.Text or Theme.Muted)
        end
        -- The mockup has no bright side indicator or selected tile.
        nav.Outline.Transparency = 1
    end

    -- Lumen owns the real page/control frames. The page frame itself has been mounted
    -- over Syllinse's original pair of panels, so use Lumen's native Open/Close behavior.
    local lumenPage = lumenPages[name]
    if lumenPage and type(lumenPage.Open) == "function" then
        pcall(function() lumenPage:Open() end)
    end
    for tabName, page in pairs(lumenPages) do
        local hostsForPage = lumenHosts[tabName]
        local selected = tabName == name
        local usingMobileHost = compactTouchLayout and hostsForPage and hostsForPage.Mobile
        if page.Frame then
            page.Frame.Visible = selected and not usingMobileHost
        end
        if hostsForPage and hostsForPage.Mobile then
            hostsForPage.Mobile.Visible = selected and compactTouchLayout
            if selected and compactTouchLayout then
                hostsForPage.Mobile.CanvasPosition = Vector2.new(0, 0)
            end
        end
    end
    local hosts = lumenHosts[name]
    if hosts then
        if hosts.Left then hosts.Left.CanvasPosition = Vector2.new(0, 0) end
        if hosts.Right then hosts.Right.CanvasPosition = Vector2.new(0, 0) end
        if hosts.Mobile then hosts.Mobile.CanvasPosition = Vector2.new(0, 0) end
    end
end

    local function applyNavHover(isHovered)
        if navReference then
            navReference.Hovering = isHovered
        end
        if navHoverState ~= isHovered then
            navHoverState = isHovered
            -- Keep the icon's subtle 1.04x hover animation. Text stays at native scale
            -- so its letterforms remain crisp; the label color still animates with hover.
            tween(hoverScale, 0.16, { Scale = isHovered and 1.04 or 1 })
        end

        -- optional terminology label react to hover, while the transparent button
        -- continues to handle clicks/taps normally.
        button.BackgroundTransparency = 1
        if isHovered then
            if item.Name ~= activeName then
                if icon then tween(icon, 0.14, { ImageColor3 = Theme.IconHover }) end

                tween(terminologyLabel, 0.14, { TextColor3 = Theme.IconIdle })
            else
                tween(terminologyLabel, 0.14, { TextColor3 = terminologyTint(Theme.IconActive, 0.76) })
            end
        end
    end

        -- MouseLeave fires as the pointer crosses the gap between the icon button
        -- and its label. Wait briefly so the independent label can claim the hover
        -- state without adding any pixels to the button's clickable hitbox.
        task.delay(0.08, function()
            if not buttonPointerInside and not labelPointerInside then
                applyNavHover(false)
            end
        end)
    end

        buttonPointerInside = true
        applyNavHover(true)
    end)
    button.MouseLeave:Connect(function()
        buttonPointerInside = false
        deferHoverExit()
    end)

        if showMenuFunctionTerminology and terminologyLabel.Visible then

            applyNavHover(true)
        end
    end)

    end)

        -- Category-prefix navigation (such as "c") clears when the user manually switches away.
        if searchBox and currentSearchMode ~= "controls" then
            searchBox.Text = ""
        end
    end)

        Hovering = false,
    }

end

local windowIsOpen = false

local windowTransitionTweens = { Scale = nil, Settle = nil, Position = nil, PositionSettle = nil, Rotation = nil, TargetPosition = nil }
local setWindowOpen

local function refreshLauncherVisibility()

-- Always animate the Syllinse window around the viewport center. The floating launcher
-- is independently draggable; its position must never determine where the main UI opens.
local function centeredWindowPosition()
    if touchInputEnabled then
        return UDim2.new(0.5, 0, 0.5, mobileSafeOffsetY)
    end
    return UDim2.fromScale(0.5, 0.5)
end

setWindowOpen = function(open)
    open = open == true
    if open == windowIsOpen then return end

    -- Cancel any in-flight animation before reversing direction. The launcher is never
    -- used as a travel destination: the shell opens and closes in place, and the launcher
    -- stays anchored exactly where the user left it.
    for _, activeTween in pairs({
        windowTransitionTweens.Scale,
        windowTransitionTweens.Settle,
        windowTransitionTweens.Position,
        windowTransitionTweens.PositionSettle,
        windowTransitionTweens.Rotation,
    }) do
        if activeTween then
            pcall(function() activeTween:Cancel() end)
        end
    end
    windowTransitionTweens.Scale = nil
    windowTransitionTweens.Settle = nil
    windowTransitionTweens.Position = nil
    windowTransitionTweens.PositionSettle = nil
    windowTransitionTweens.Rotation = nil

    windowIsOpen = open
    windowTransitionId += 1
    local thisTransition = windowTransitionId

    local targetPosition
    if windowHasCustomPosition then
        targetPosition = windowTransitionTweens.TargetPosition or root.Position
    else
        targetPosition = centeredWindowPosition()
    end
    windowTransitionTweens.TargetPosition = targetPosition

    -- Always keep the shell anchored at its final position. This intentionally removes
    -- the launcher-to-window travel that made the UI shoot upward and appear to hide.
    root.Position = targetPosition
    root.Rotation = 0

    if open then
        local wasVisible = root.Visible
        if baseWindowSize then root.Size = baseWindowSize end

        if not wasVisible then
            -- A compact, centered start gives a premium reveal without moving the window.
            windowMotionFactor.Value = 0.88
        end

        root.BackgroundColor3 = Theme.Root
        root.BackgroundTransparency = 0
        root.ClipsDescendants = true
        root.Visible = true
        motionBody.GroupTransparency = 0
        motionBody.Position = UDim2.fromOffset(0, 0)
        refreshLauncherVisibility()

        -- One-piece reveal: the entire opaque shell expands, then gently settles.
        local openInfo = TweenInfo.new(0.19, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
        local settleInfo = TweenInfo.new(0.075, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
        windowTransitionTweens.Scale = TweenService:Create(windowMotionFactor, openInfo, {
            Value = 1.018,
        })
        windowTransitionTweens.Scale.Completed:Connect(function(playbackState)
            if playbackState ~= Enum.PlaybackState.Completed then return end
            if thisTransition ~= windowTransitionId or not windowIsOpen then return end

            windowTransitionTweens.Settle = TweenService:Create(windowMotionFactor, settleInfo, {
                Value = 1,
            })
            windowTransitionTweens.Settle.Completed:Connect(function(settleState)
                if settleState ~= Enum.PlaybackState.Completed then return end
                if thisTransition ~= windowTransitionId or not windowIsOpen then return end
                windowTransitionTweens.Scale = nil
                windowTransitionTweens.Settle = nil
            end)
            windowTransitionTweens.Settle:Play()
        end)
        windowTransitionTweens.Scale:Play()
    else
        -- Preserve Lumen overlays so keybind lists, watermarks, and notifications continue working.
        if Lumen and Lumen.OpenPopup and type(Lumen.OpenPopup.Close) == "function" then
            pcall(function() Lumen.OpenPopup.Close() end)
        end
        if Lumen and Lumen._Instance then Lumen._Instance.Enabled = true end
        if Lumen and Lumen._Overlay then Lumen._Overlay.Enabled = true end

        -- Fast, centered close. No translation, no partial-content fade, and no outline flash.
        local closeInfo = TweenInfo.new(0.12, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
        windowTransitionTweens.Scale = TweenService:Create(windowMotionFactor, closeInfo, {
            Value = 0.88,
        })
        windowTransitionTweens.Scale.Completed:Connect(function(playbackState)
            if playbackState ~= Enum.PlaybackState.Completed then return end
            if thisTransition ~= windowTransitionId or windowIsOpen then return end

            root.Visible = false
            root.BackgroundColor3 = Theme.Root
            root.BackgroundTransparency = 0
            root.Rotation = 0
            motionBody.GroupTransparency = 0
            motionBody.Position = UDim2.fromOffset(0, 0)
            root.Position = targetPosition
            windowTransitionTweens.TargetPosition = targetPosition
            windowMotionFactor.Value = 1
            uiScale.Scale = responsiveScaleValue
            windowTransitionTweens.Scale = nil
            windowTransitionTweens.Settle = nil
            windowTransitionTweens.Position = nil
            windowTransitionTweens.PositionSettle = nil
            windowTransitionTweens.Rotation = nil
            refreshLauncherVisibility()
        end)
        windowTransitionTweens.Scale:Play()
    end
end

        -- InputObject.Position is a Vector3 in Roblox; convert screen X/Y to Vector2
        -- before comparing it with GuiObject.AbsolutePosition (a Vector2).
        local inputPosition = Vector2.new(input.Position.X, input.Position.Y)

UserInputService.InputChanged:Connect(function(input)
    if not draggingLauncher then return end
    if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local inputPosition = Vector2.new(input.Position.X, input.Position.Y)
    if (inputPosition - launcherDragStart).Magnitude > 3 then
        launcherMoved = true
    end
    if not launcherMoved then return end

    local point = inputPosition - launcherGrab - gui.AbsolutePosition
    launcher.AnchorPoint = Vector2.new(0, 0)
    launcher.Position = UDim2.fromOffset(
        math.clamp(point.X, 0, math.max(gui.AbsoluteSize.X - launcher.AbsoluteSize.X, 0)),
        math.clamp(point.Y, 0, math.max(gui.AbsoluteSize.Y - launcher.AbsoluteSize.Y, 0))
    )
    syncLauncherGlow()
end)
UserInputService.InputEnded:Connect(function(input)
    if not draggingLauncher then return end
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    draggingLauncher = false
    if not launcherMoved then
        setWindowOpen(not windowIsOpen)
    end
end)

-- Drag the shell only from the dedicated four-way icon.
-- Keep the entire shell inside the top/bottom and right edges, while allowing it to
-- move partly beyond the left edge as long as the top-right drag icon stays reachable.
local dragging = false
local dragInput
local dragStart
local startPosition

local function beginShellDrag(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    dragging = true
    -- Store the active pointer mode in dragInput. Mouse drags follow global mouse-move
    -- events; touch drags follow only the touch that began on the drag handle.
    dragInput = input.UserInputType == Enum.UserInputType.Touch and input or Enum.UserInputType.MouseButton1
    dragStart = input.Position
    startPosition = root.Position
    input.Changed:Connect(function()
        if input.UserInputState == Enum.UserInputState.End then
            dragging = false
            dragInput = nil
        end
    end)
end

-- Only the dedicated four-way drag icon can move the Syllinse shell.
dragButton.InputBegan:Connect(beginShellDrag)

UserInputService.InputChanged:Connect(function(input)
    local pointerMoved = dragging and (
        (dragInput == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseMovement)
        or (dragInput ~= Enum.UserInputType.MouseButton1 and dragInput == input and input.UserInputType == Enum.UserInputType.Touch)
    )
    if pointerMoved then
        local pointerDelta = input.Position - dragStart
        if pointerDelta.Magnitude > 3 then
            windowHasCustomPosition = true
        end

        local camera = Workspace.CurrentCamera
        local viewport = camera and camera.ViewportSize or gui.AbsoluteSize
        local shellSize = root.AbsoluteSize
        local dragSize = dragButton.AbsoluteSize
        local centerX = viewport.X * startPosition.X.Scale + startPosition.X.Offset + pointerDelta.X
        local centerY = viewport.Y * startPosition.Y.Scale + startPosition.Y.Offset + pointerDelta.Y

        -- The left edge can leave the screen, but the drag icon itself must remain visible.
        local minCenterX = 8 - shellSize.X / 2 + 6 + dragSize.X
        -- Keep the right edge of the shell on-screen so it cannot hide the drag icon.
        local maxCenterX = viewport.X - shellSize.X / 2 - 8
        if minCenterX > maxCenterX then
            maxCenterX = viewport.X - shellSize.X / 2 + 2
            minCenterX = math.min(minCenterX, maxCenterX)
        end
        centerX = math.clamp(centerX, minCenterX, maxCenterX)

        -- Top matches the right side: keep the complete shell inside the top margin.
        -- Bottom remains flexible, but the top-right drag handle must stay reachable.
        local dragOffsetY = dragButton.AbsolutePosition.Y - root.AbsolutePosition.Y
        local minCenterY = 8 + shellSize.Y / 2
        local maxCenterY = viewport.Y - 8 - dragOffsetY - dragSize.Y + shellSize.Y / 2
        if maxCenterY < minCenterY then
            centerY = minCenterY
        else
            centerY = math.clamp(centerY, minCenterY, maxCenterY)
        end

        root.Position = UDim2.new(
            startPosition.X.Scale,
            centerX - viewport.X * startPosition.X.Scale,
            startPosition.Y.Scale,
            centerY - viewport.Y * startPosition.Y.Scale
        )
        windowTransitionTweens.TargetPosition = root.Position

        -- Rebase after each update. This prevents edge-clamping from feeling sticky:
        -- when the pointer reverses direction, the shell moves immediately.
        dragStart = input.Position
        startPosition = root.Position
    end
end)

UserInputService.InputEnded:Connect(function(input)

        or (dragInput ~= Enum.UserInputType.MouseButton1 and input == dragInput and input.UserInputType == Enum.UserInputType.Touch) then
        dragging = false
        dragInput = nil
    end
end)

    if not instance or not instance:IsA("GuiObject") then
        return

    if not popupScale then
        popupScale = Instance.new("UIScale")
        popupScale.Name = "SyllinseDropdownScale"
        popupScale.Parent = instance

end

SyllinseRuntime.setNotificationPosition = function(position)
    if type(position) ~= "string" then return false end
    if not table.find({ "Top Left", "Top Right", "Bottom Left", "Bottom Right" }, position) then
        return false
    end

    SyllinseRuntime.notificationPosition = position
    local host = SyllinseRuntime.getNotificationHost()
    local left = position == "Top Left" or position == "Bottom Left"
    local bottom = position == "Bottom Left" or position == "Bottom Right"
    local inset = SyllinseRuntime.getNotificationInset()
    host.Size = UDim2.new(0, SyllinseRuntime.getNotificationWidth(), 1, -(inset * 2))
    host.AnchorPoint = Vector2.new(left and 0 or 1, bottom and 1 or 0)
    host.Position = UDim2.new(left and 0 or 1, left and inset or -inset, bottom and 1 or 0, bottom and -inset or inset)
    if SyllinseRuntime.notificationLayout then
        SyllinseRuntime.notificationLayout.HorizontalAlignment = left and Enum.HorizontalAlignment.Left or Enum.HorizontalAlignment.Right
        SyllinseRuntime.notificationLayout.VerticalAlignment = bottom and Enum.VerticalAlignment.Bottom or Enum.VerticalAlignment.Top
    end
    return true
end

-- Lumen's compact label rows are 14 px tall by default. When a touch control grows,
-- expand its owning row as well so larger controls never collide with the next row.
Lumen._SyllinseExpandMobileControlRow = function(control, targetHeight)
    if not (touchInputEnabled and not UserInputService.KeyboardEnabled) or not control then return end
    local ancestor = control.Parent
    while ancestor and ancestor ~= root do
        if ancestor.Name == "LabelFrame" then
            local size = ancestor.Size
            local scaledTargetHeight = math.ceil((targetHeight or 48) / math.max(responsiveScaleValue, 0.25))
            if size.Y.Scale == 0 and size.Y.Offset < scaledTargetHeight then
                ancestor.Size = UDim2.new(size.X.Scale, size.X.Offset, 0, scaledTargetHeight)
            end
            ancestor.ClipsDescendants = false
            -- Keep both halves centered in the taller row rather than pinning icons to the top.
            local leftContent = ancestor:FindFirstChild("LeftContent")
            if leftContent then
                leftContent.AnchorPoint = Vector2.new(0, 0.5)
                leftContent.Position = UDim2.new(0, 0, 0.5, 0)
            end
            local rightContent = ancestor:FindFirstChild("RightContent")
            if rightContent then
                rightContent.AnchorPoint = Vector2.new(1, 0.5)
                rightContent.Position = UDim2.new(1, 0, 0.5, 0)
            end
            return ancestor
        end
        ancestor = ancestor.Parent
    end
end

SyllinseRuntime.attachFeatureNotificationBell = function(label, toggleControl, featureFlag)
    if not label or not label.RightContent or not toggleControl or not featureFlag then return end
    local rightContent = label.RightContent
    if not rightContent.Parent then return end

    -- The bell belongs only beside a real keybind/settings gear. Some toggles do not
    -- have a keybind control, and must not get an orphan notification icon.
    local settingsIcon = rightContent:FindFirstChild("Settings")
    if not settingsIcon or not settingsIcon:IsA("GuiObject") then return end
    if (touchInputEnabled and not UserInputService.KeyboardEnabled) then
        -- Keep the gear artwork at its original aspect ratio. The mobile styling pass
        -- enlarges its hit target later and renders the artwork in a separate fitted label.
        Lumen._SyllinseExpandMobileControlRow(settingsIcon, 40)
    end

    local existingBell = rightContent:FindFirstChild("FeatureNotificationBell_" .. featureFlag)
    if existingBell then
        if (touchInputEnabled and not UserInputService.KeyboardEnabled) then
            Lumen._SyllinseExpandMobileControlRow(existingBell, 40)
        end
        return existingBell
    end

    if SyllinseRuntime.featureNotificationSettings[featureFlag] == nil then
        SyllinseRuntime.featureNotificationSettings[featureFlag] = false
    end
    toggleControl.NotifyEnabled = SyllinseRuntime.featureNotificationSettings[featureFlag] == true

    local bell = Instance.new("ImageButton")
    bell.Name = "FeatureNotificationBell_" .. featureFlag
    -- Keep the underlying artwork at its original size. On phones, the styling pass
    -- adds a larger transparent hit target and a separate aspect-preserving visual.
    bell.Size = UDim2.fromOffset(14, 14)
    bell.BackgroundTransparency = 1
    bell.BackgroundColor3 = Theme.SurfaceHover
    bell.BorderSizePixel = 0
    bell.AutoButtonColor = false
    bell.Image = "rbxassetid://10709775704"
    bell.ImageColor3 = SyllinseRuntime.featureNotificationSettings[featureFlag] and Theme.Text or Theme.Dim
    bell.ScaleType = Enum.ScaleType.Fit
    bell.LayoutOrder = 1
    bell.ZIndex = 8
    bell.Parent = rightContent
    Lumen._SyllinseExpandMobileControlRow(bell, 40)

    -- Keep the bell immediately LEFT of the keybind/settings gear. Other inline
    -- controls (such as a color picker) stay after the bell/gear pair.
    for _, sibling in ipairs(rightContent:GetChildren()) do
        if sibling:IsA("GuiObject") and sibling ~= bell then
            if sibling == settingsIcon then
                sibling.LayoutOrder = 2
            else
                sibling.LayoutOrder = 3 + math.max(0, tonumber(sibling.LayoutOrder) or 0)
            end
        end
    end
    bell.LayoutOrder = 1

    bell:SetAttribute("FeatureFlag", featureFlag)
    bell:SetAttribute("Tooltip", "Toggle notifications for " .. tostring(label.Text or featureFlag))
    addCorner(bell, (touchInputEnabled and not UserInputService.KeyboardEnabled) and 7 or 4)

    local activeDot = Instance.new("Frame")
    activeDot.Name = "NotificationEnabledDot"
    activeDot.Size = (touchInputEnabled and not UserInputService.KeyboardEnabled) and UDim2.fromOffset(5, 5) or UDim2.fromOffset(4, 4)
    activeDot.AnchorPoint = Vector2.new(1, 0)
    activeDot.Position = UDim2.new(1, -1, 0, 0)
    activeDot.BackgroundColor3 = Theme.Text
    activeDot.BorderSizePixel = 0
    activeDot.Visible = SyllinseRuntime.featureNotificationSettings[featureFlag] == true
    activeDot.ZIndex = 9
    activeDot.Parent = bell
    addCorner(activeDot, 4)

    local function refreshBell()
        local enabled = SyllinseRuntime.featureNotificationSettings[featureFlag] == true
        toggleControl.NotifyEnabled = enabled
        SyllinseRuntime.featureSettings.FeatureNotifications[featureFlag] = enabled
        bell.ImageColor3 = enabled and Theme.Text or Theme.Dim
        activeDot.Visible = enabled
    end

    bell.MouseEnter:Connect(function()
        bell.BackgroundTransparency = 0.18
        bell.ImageColor3 = Theme.Text
    end)
    bell.MouseLeave:Connect(function()
        bell.BackgroundTransparency = 1
        bell.ImageColor3 = SyllinseRuntime.featureNotificationSettings[featureFlag] and Theme.Text or Theme.Dim
    end)
    bell.Activated:Connect(function()
        SyllinseRuntime.featureNotificationSettings[featureFlag] = not (SyllinseRuntime.featureNotificationSettings[featureFlag] == true)
        refreshBell()
        -- Deliberately do not toast here: clicking the bell changes preference only.
    end)

    refreshBell()
    return bell
end

-- Mobile touch integrations adapted from MacLib while keeping Syllinse's existing visuals.
-- A vertical swipe through a scroll container should scroll, not accidentally toggle a row.
SyllinseRuntime.mobileScrollGuards = {}
SyllinseRuntime.mobileScrollDispatcherInstalled = false
SyllinseRuntime.mobileScrollInputChangedConnection = nil
SyllinseRuntime.mobileScrollInputEndedConnection = nil

SyllinseRuntime.setGuardedButtonInteractable = function(state, button, desiredState)
    if not button or not button.Parent or not button:IsA("GuiButton") then return end
    if desiredState then
        local original = state.OriginalInteractable[button]
        if original ~= nil then
            pcall(function() button.Interactable = original end)
        end
        state.OriginalInteractable[button] = nil
    else
        if state.OriginalInteractable[button] == nil then
            local ok, original = pcall(function() return button.Interactable end)
            if not ok then return end
            state.OriginalInteractable[button] = original
        end
        pcall(function() button.Interactable = false end)
    end
end

SyllinseRuntime.setMobileScrollState = function(state, scrolling)
    if not state or state.Destroyed then return end
    state.Scrolling = scrolling == true
    pcall(function()
        state.Frame:SetAttribute("_mobileScrolling", state.Scrolling) -- MacLib-compatible marker
        state.Frame:SetAttribute("_syllinseMobileScrolling", state.Scrolling)
    end)
    if state.Scrolling then
        for _, descendant in ipairs(state.Frame:GetDescendants()) do
            if descendant:IsA("GuiButton") then
                SyllinseRuntime.setGuardedButtonInteractable(state, descendant, false)
            end
        end
    else
        for button, original in pairs(state.OriginalInteractable) do
            if button and button.Parent then
                pcall(function() button.Interactable = original end)
            end
        end
        table.clear(state.OriginalInteractable)
    end
end

SyllinseRuntime.bindMobileScrollGuard = function(scrollFrame)
    if not touchInputEnabled or not scrollFrame or not scrollFrame:IsA("ScrollingFrame") then return end
    if scrollFrame:GetAttribute("_syllinseTouchGuardBound") then return end
    scrollFrame:SetAttribute("_syllinseTouchGuardBound", true)
    scrollFrame:SetAttribute("_mobileScrolling", false)
    scrollFrame:SetAttribute("_syllinseMobileScrolling", false)
    scrollFrame.Active = true

    local state = {
        Frame = scrollFrame,
        ActiveInput = nil,
        StartPosition = nil,
        Scrolling = false,
        OriginalInteractable = {},
        Destroyed = false,
    }
    table.insert(SyllinseRuntime.mobileScrollGuards, state)

    scrollFrame.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.Touch then return end
        state.ActiveInput = input
        state.StartPosition = Vector2.new(input.Position.X, input.Position.Y)
        if state.Scrolling then SyllinseRuntime.setMobileScrollState(state, false) end
    end)
    scrollFrame.InputChanged:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.Touch or not state.ActiveInput then return end
        local point = Vector2.new(input.Position.X, input.Position.Y)
        local delta = point - (state.StartPosition or point)
        if math.abs(delta.Y) > 8 and math.abs(delta.Y) > math.abs(delta.X) then
            SyllinseRuntime.setMobileScrollState(state, true)
        end
    end)
    scrollFrame.DescendantAdded:Connect(function(descendant)
        if state.Scrolling and descendant:IsA("GuiButton") then
            SyllinseRuntime.setGuardedButtonInteractable(state, descendant, false)
        end
    end)
    scrollFrame.Destroying:Connect(function()
        SyllinseRuntime.setMobileScrollState(state, false)
        state.Destroyed = true
        for index = #SyllinseRuntime.mobileScrollGuards, 1, -1 do
            if SyllinseRuntime.mobileScrollGuards[index] == state then
                table.remove(SyllinseRuntime.mobileScrollGuards, index)
                break
            end
        end
        if #SyllinseRuntime.mobileScrollGuards == 0 then
            if SyllinseRuntime.mobileScrollInputChangedConnection then SyllinseRuntime.mobileScrollInputChangedConnection:Disconnect() end
            if SyllinseRuntime.mobileScrollInputEndedConnection then SyllinseRuntime.mobileScrollInputEndedConnection:Disconnect() end
            SyllinseRuntime.mobileScrollInputChangedConnection = nil
            SyllinseRuntime.mobileScrollInputEndedConnection = nil
            SyllinseRuntime.mobileScrollDispatcherInstalled = false
        end
    end)

    if not SyllinseRuntime.mobileScrollDispatcherInstalled then
        SyllinseRuntime.mobileScrollDispatcherInstalled = true
        SyllinseRuntime.mobileScrollInputChangedConnection = UserInputService.InputChanged:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.Touch then return end
            local point = Vector2.new(input.Position.X, input.Position.Y)
            for _, activeState in ipairs(SyllinseRuntime.mobileScrollGuards) do
                if not activeState.Destroyed and activeState.ActiveInput == input then
                    local delta = point - (activeState.StartPosition or point)
                    if math.abs(delta.Y) > 8 and math.abs(delta.Y) > math.abs(delta.X) then
                        SyllinseRuntime.setMobileScrollState(activeState, true)
                    end
                end
            end
        end)
        SyllinseRuntime.mobileScrollInputEndedConnection = UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.Touch then return end
            for _, activeState in ipairs(SyllinseRuntime.mobileScrollGuards) do
                if activeState.ActiveInput
                    and (input == activeState.ActiveInput or activeState.ActiveInput.UserInputState == Enum.UserInputState.End) then
                    local endedInput = activeState.ActiveInput
                    task.delay(0.05, function()
                        if not activeState.Destroyed and activeState.ActiveInput == endedInput then
                            SyllinseRuntime.setMobileScrollState(activeState, false)
                            activeState.ActiveInput = nil
                            activeState.StartPosition = nil
                        end
                    end)
                end
            end
        end)
    end
end

SyllinseRuntime.clampMobileActionButtonsToViewport = function()
    task.defer(function()
        if not mobileActionGui or not mobileActionGui.Parent then return end
        local camera = Workspace.CurrentCamera
        local viewport = camera and camera.ViewportSize or Vector2.new(390, 844)
        for _, entry in ipairs(SyllinseRuntime.mobileActionButtons) do
            local button = entry.Button
            if button and button.Parent then
                local position = button.AbsolutePosition
                local size = button.AbsoluteSize
                local left = math.clamp(position.X, 4, math.max(4, viewport.X - size.X - 4))
                local top = math.clamp(position.Y, 4, math.max(4, viewport.Y - size.Y - 4))
                button.AnchorPoint = Vector2.new(0, 0)
                button.Position = UDim2.fromOffset(left, top)
            end
        end
    end)
end

    end

        SyllinseRuntime.watchMobileActionCamera(Workspace.CurrentCamera)

        if SyllinseRuntime.actionCurrentCameraConnection then SyllinseRuntime.actionCurrentCameraConnection:Disconnect() end

    end)
end

-- Lumen's provided Toggle:Set implementation treats a false argument as a toggle
-- request. Normalize that edge case once per toggle so false defaults and config loads
-- remain accurate while retaining Lumen's native animation/callback/flag behavior.
SyllinseRuntime.addToggle = function(label, options)
    local opts = options or {}
    local requestedState = opts.State == true
    local featureFlag = type(opts.Flag) == "string" and opts.Flag or nil
    local hasFeatureNotifications = featureFlag ~= nil and SyllinseRuntime.FEATURE_NOTIFICATION_FLAGS[featureFlag] == true
    if hasFeatureNotifications then
        if SyllinseRuntime.featureNotificationSettings[featureFlag] == nil then
            SyllinseRuntime.featureNotificationSettings[featureFlag] = false
        end
        opts.NotifyEnabled = SyllinseRuntime.featureNotificationSettings[featureFlag] == true
    else
        -- Do not let unrelated controls emit native toggle toasts.
        opts.NotifyEnabled = false
    end
    local control = label:Toggle(opts)
    if hasFeatureNotifications then
        task.defer(function()
            SyllinseRuntime.attachFeatureNotificationBell(label, control, featureFlag)
        end)
    end
    local originalSet = control.Set
    local currentState = control.State == true

    -- Keep Lumen's native toggle behavior and geometry, but use a visible charcoal
    -- off track. Lumen's internal click handler tweens the track back to near-black,
    -- so the style is also re-applied after native activation/state animations.
    local function applySyllinseToggleStyle(isOn)
        local frame = control and control.Frame
        if not frame or not frame.Parent then return end
        local track
        for _, object in ipairs(frame:GetDescendants()) do
            if object.Name == "Toggle" and object:IsA("GuiButton") then
                track = object
                break
            end
        end
        if not track then return end

        -- Keep one consistent base switch size on each device type. A child UIScale
        -- counteracts the shell's global scaling so the switch stays easy to use even
        -- when the user lowers the overall UI-size slider.
        local touchFirst = touchInputEnabled and not UserInputService.KeyboardEnabled
        local baseSwitchWidth = touchFirst and 50 or 34
        local baseSwitchHeight = touchFirst and 30 or 18
        track:SetAttribute("SyllinseBaseSwitchWidth", baseSwitchWidth)
        track:SetAttribute("SyllinseBaseSwitchHeight", baseSwitchHeight)
        track.Size = UDim2.fromOffset(baseSwitchWidth, baseSwitchHeight)
        local switchScale = track:FindFirstChild("SyllinseSwitchScaleCompensation")
        if not switchScale then
            switchScale = Instance.new("UIScale")
            switchScale.Name = "SyllinseSwitchScaleCompensation"
            switchScale.Parent = track
        end
        switchScale.Scale = 1 / math.max(responsiveScaleValue, 0.25)

        -- Reserve enough row height for the switch at the current UI scale. Keep
        -- the original row height so the layout can recover cleanly when scale changes.
        local switchRow = track.Parent
        while switchRow and switchRow ~= root do
            if switchRow.Name == "LabelFrame" then
                if switchRow:GetAttribute("SyllinseToggleBaseRowHeight") == nil then
                    switchRow:SetAttribute("SyllinseToggleBaseRowHeight", switchRow.Size.Y.Offset)
                end
                local baseHeight = switchRow:GetAttribute("SyllinseToggleBaseRowHeight") or switchRow.Size.Y.Offset
                local minimumPhysicalHeight = touchFirst and 48 or 24
                local targetHeight = math.max(baseHeight, minimumPhysicalHeight / math.max(responsiveScaleValue, 0.25))
                if switchRow.Size.Y.Scale == 0 then
                    switchRow.Size = UDim2.new(switchRow.Size.X.Scale, switchRow.Size.X.Offset, 0, math.ceil(targetHeight))
                    switchRow.ClipsDescendants = false
                end
                break
            end
            switchRow = switchRow.Parent
        end

        track.BackgroundColor3 = isOn and Theme.PanelTitle or Theme.ToggleOffTrack
        track.BackgroundTransparency = 0
        track.AutoButtonColor = false
        track.ClipsDescendants = true

        local trackCorner = track:FindFirstChildOfClass("UICorner")
        if trackCorner then
            trackCorner.CornerRadius = UDim.new(1, 0)
        else
            addCorner(track, 9)
        end

        -- Lumen calls the switch thumb "Indicator". Scale and reposition it with the
        -- larger mobile switch so the thumb remains centered and easy to see.
        if (touchInputEnabled and not UserInputService.KeyboardEnabled) then
            local indicator = track:FindFirstChild("Indicator")
            if indicator and indicator:IsA("GuiObject") then
                indicator.Size = UDim2.fromOffset(20, 20)
                indicator.BackgroundColor3 = isOn and Theme.Text or Theme.ToggleOffThumb
                indicator.BackgroundTransparency = 0
                if isOn then
                    indicator.AnchorPoint = Vector2.new(1, 0.5)
                    indicator.Position = UDim2.new(1, -5, 0.5, 0)
                else
                    indicator.AnchorPoint = Vector2.new(0, 0.5)
                    indicator.Position = UDim2.new(0, 5, 0.5, 0)
                end
            end
        end

        local trackStroke = track:FindFirstChildOfClass("UIStroke")
        if not trackStroke then
            trackStroke = addStroke(track, Theme.Stroke, 0.12, 1)
        end
        trackStroke.Color = isOn and Theme.PanelTitle or Theme.Stroke
        trackStroke.Transparency = isOn and 0.22 or 0.08

        -- Lumen's own click handler can overwrite BackgroundColor3 after the wrapper's
        -- Set() call, and normal clicks do not necessarily call the public Set method.
        -- Bind the native Activated event once, then repaint after its state/tween work.
        if track:GetAttribute("SyllinseToggleStyleBound") ~= true then
            track:SetAttribute("SyllinseToggleStyleBound", true)
            track.Activated:Connect(function()
                for _, delaySeconds in ipairs({0, 0.04, 0.10, 0.20, 0.34}) do
                    task.delay(delaySeconds, function()
                        if not control or not control.Frame or not control.Frame.Parent then return end
                        if type(control.State) == "boolean" then
                            currentState = control.State
                        end
                        applySyllinseToggleStyle(currentState)
                    end)
                end
            end)
        end

        for _, part in ipairs(track:GetDescendants()) do
            local partName = string.lower(part.Name or "")
            local isThumb = partName:find("circle", 1, true)
                or partName:find("knob", 1, true)
                or partName:find("handle", 1, true)
            if isThumb and part:IsA("GuiObject") then
                if part:IsA("ImageLabel") or part:IsA("ImageButton") then
                    part.ImageColor3 = isOn and Theme.Text or Theme.ToggleOffThumb
                    part.ImageTransparency = 0
                else
                    part.BackgroundColor3 = isOn and Theme.Text or Theme.ToggleOffThumb
                    part.BackgroundTransparency = 0
                end
                local thumbCorner = part:FindFirstChildOfClass("UICorner")
                if thumbCorner then
                    thumbCorner.CornerRadius = UDim.new(1, 0)
                elseif part:IsA("Frame") or part:IsA("ImageLabel") then
                    addCorner(part, 20)
                end
            end
        end
    end
    task.defer(function() applySyllinseToggleStyle(currentState) end)

    control.Set = function(state, silent)
        if state == nil then state = not currentState end
        state = state == true
        if state == currentState then
            return currentState
        end
        currentState = state
        originalSet(state, silent)
        task.defer(function()
            if type(control.State) == "boolean" then currentState = control.State end
            applySyllinseToggleStyle(currentState)
        end)
        task.delay(0.20, function()
            if control and control.Frame and control.Frame.Parent then
                if type(control.State) == "boolean" then currentState = control.State end
                applySyllinseToggleStyle(currentState)
            end
        end)
        SyllinseRuntime.syncMobileActionButtons(featureFlag, currentState)
        if control.Flag and Lumen.Flags[control.Flag] then
            Lumen.Flags[control.Flag].Value = currentState
            Lumen.Flags[control.Flag].Set = control.Set
        end
        if hasFeatureNotifications and not silent and SyllinseRuntime.featureNotificationSettings[featureFlag] == true then
            SyllinseRuntime.notify(opts.Name or featureFlag, currentState and "Enabled" or "Disabled",
                currentState and "Success" or "Info", opts.NotificationDuration or 2.5)
        end
        return currentState
    end

    if control.Flag then
        Lumen.Flags[control.Flag] = Lumen.Flags[control.Flag] or {}
        Lumen.Flags[control.Flag].Set = control.Set
    end
    if currentState ~= requestedState then
        control.Set(requestedState, true)
    end
    return control
end

SyllinseRuntime.createMobileActionButton = function(keybindFlag, featureFlag, callback, initialState)
    if not mobileActionGui or type(callback) ~= "function" then return nil end
    local index = #SyllinseRuntime.mobileActionButtons + 1
    local label = SyllinseRuntime.MOBILE_ACTION_LABELS[keybindFlag] or tostring(keybindFlag or "ACTION")

    local viewportCamera = Workspace.CurrentCamera
    local viewportHeight = viewportCamera and viewportCamera.ViewportSize.Y or 844
    local buttonHeight = 34
    local stackTop = math.min(viewportHeight * 0.34, math.max(8, viewportHeight - (#SyllinseRuntime.mobileActionButtons + 1) * 42 - 8))
    local labelWidths = { AIM = 64, SILENT = 78, TRIGGER = 84, ["ANTI-AIM"] = 92, FLY = 60 }
    local button = Instance.new("TextButton")
    button.Name = "SyllinseMobileAction_" .. tostring(keybindFlag or index)
    button.AnchorPoint = Vector2.new(1, 0)
    button.Position = UDim2.new(1, -12, 0, stackTop + (index - 1) * 42)
    button.Size = UDim2.fromOffset(labelWidths[label] or 78, buttonHeight)
    button.BackgroundColor3 = Theme.Surface
    button.BackgroundTransparency = 0.03
    button.BorderSizePixel = 0
    button.AutoButtonColor = false
    button.FontFace = FONT_SEMIBOLD
    button.Text = label
    button.TextColor3 = Theme.PanelText
    button.TextSize = 12
    button.TextTruncate = Enum.TextTruncate.AtEnd
    button.TextXAlignment = Enum.TextXAlignment.Center
    button.TextYAlignment = Enum.TextYAlignment.Center
    button.Visible = false
    button.Active = true
    button.Selectable = false
    button.ZIndex = 250
    button.Parent = mobileActionGui
    addCorner(button, 10)
    local stroke = addStroke(button, Theme.Stroke, 0.08, 1)

    local entry = {
        KeybindFlag = keybindFlag,
        FeatureFlag = featureFlag,
        Callback = callback,
        Button = button,
        Stroke = stroke,
        Active = initialState == true,
        ManualVisible = false,
        DragInput = nil,
    }
    table.insert(SyllinseRuntime.mobileActionButtons, entry)
    SyllinseRuntime.refreshMobileActionButton(entry, initialState)

    -- Same touch/mouse drag-and-release pattern as MacLib's mobile keybind API.
    -- A short tap toggles the linked feature; dragging only repositions the button.
    local gesture
    button.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.Touch
            and input.UserInputType ~= Enum.UserInputType.MouseButton1 then
            return
        end
        gesture = {
            Input = input,
            StartPoint = Vector2.new(input.Position.X, input.Position.Y),
            StartPosition = button.AbsolutePosition,
            Dragged = false,
        }
    end)

    -- Scope movement updates to this button, matching MacLib's per-keybind drag handler.
    entry.InputChangedConnection = button.InputChanged:Connect(function(input)
        if not gesture then return end
        local isTouchMove = gesture.Input.UserInputType == Enum.UserInputType.Touch
            and input.UserInputType == Enum.UserInputType.Touch
        local isMouseMove = gesture.Input.UserInputType == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseMovement
        if not isTouchMove and not isMouseMove then return end

        local point = Vector2.new(input.Position.X, input.Position.Y)
        local delta = point - gesture.StartPoint
        if delta.Magnitude > 8 then gesture.Dragged = true end
        if not gesture.Dragged then return end

        local camera = Workspace.CurrentCamera
        local viewport = camera and camera.ViewportSize or Vector2.new(390, 844)
        local left = math.clamp(gesture.StartPosition.X + delta.X, 4, math.max(4, viewport.X - button.AbsoluteSize.X - 4))
        local top = math.clamp(gesture.StartPosition.Y + delta.Y, 4, math.max(4, viewport.Y - button.AbsoluteSize.Y - 4))
        button.AnchorPoint = Vector2.new(0, 0)
        button.Position = UDim2.fromOffset(left, top)
    end)

    entry.InputEndedConnection = UserInputService.InputEnded:Connect(function(input)
        if not gesture then return end
        local isTouchEnd = gesture.Input.UserInputType == Enum.UserInputType.Touch
            and input.UserInputType == Enum.UserInputType.Touch
        local isMouseEnd = gesture.Input.UserInputType == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseButton1
        if not isTouchEnd and not isMouseEnd then return end

        local completedGesture = gesture
        gesture = nil
        if not completedGesture.Dragged then
            local nextState = not entry.Active
            local ok, err = pcall(callback, nextState)
            if not ok then warn("[Syllinse] Mobile action failed: " .. tostring(err)) end
            -- If the callback is not linked to a toggle, still show the local action state.
            SyllinseRuntime.refreshMobileActionButton(entry, nextState)
        end
    end)

    return entry
end

SyllinseRuntime.attachKeybindSetting = function(label, options, settingName, linkedToggleControl)
    -- Lumen's Keybind callback reports the active state, but does not automatically
    -- toggle the feature's separate main switch. Link those two controls so a hotkey
    -- actually updates the same state as clicking the feature toggle.
    local keybindOptions = options or {}
    local originalCallback = keybindOptions.Callback
    if linkedToggleControl then
        keybindOptions.Callback = function(active)
            if type(originalCallback) == "function" then
                originalCallback(active)
            end
            if linkedToggleControl and type(linkedToggleControl.Set) == "function" then
                -- Use the explicit active state (not Set() with no argument), so
                -- Hold mode turns off on release and Toggle mode follows its key state.
                linkedToggleControl.Set(active)
            end
        end
    end

    local control = label:Keybind(keybindOptions)
    if settingName then
        SyllinseRuntime.featureSettings[settingName] = control.Key
        local originalSet = control.Set
        control.Set = function(key)
            originalSet(key)
            SyllinseRuntime.featureSettings[settingName] = key
            if control.Flag and Lumen.Flags[control.Flag] then
                Lumen.Flags[control.Flag].Value = key
                Lumen.Flags[control.Flag].Set = control.Set
            end
        end
        if control.Flag and Lumen.Flags[control.Flag] then
            Lumen.Flags[control.Flag].Set = control.Set
        end
    end

    if touchInputEnabled and linkedToggleControl and type(keybindOptions.Callback) == "function" then
        local initialState = linkedToggleControl.State == true
        SyllinseRuntime.createMobileActionButton(
            settingName or keybindOptions.Flag,
            linkedToggleControl.Flag,
            keybindOptions.Callback,
            initialState
        )
    end

    -- Match MacLib's per-keybind visibility API while leaving Lumen's native keybind control intact.
    control.SetMobileButtonVisibility = function(selfOrState, state)
        if selfOrState == control then
            return SyllinseRuntime.setMobileActionButtonVisibility(settingName or keybindOptions.Flag, state)
        end
        return SyllinseRuntime.setMobileActionButtonVisibility(settingName or keybindOptions.Flag, selfOrState)
    end
    return control
end

-- Bind MacLib-style scroll guards to the visible outer scroll containers in Lumen's
-- separate popup layers (for example, dropdown option lists). Avoid nested guards.
SyllinseRuntime.watchOuterMobileScrollFrames = function(layer)
    if not touchInputEnabled or not layer then return end
    local function consider(instance)
        if not instance:IsA("ScrollingFrame") then return end
        local parent = instance.Parent
        while parent and parent ~= layer do
            if parent:IsA("ScrollingFrame") then return end
            parent = parent.Parent
        end
        SyllinseRuntime.bindMobileScrollGuard(instance)
    end
    for _, descendant in ipairs(layer:GetDescendants()) do consider(descendant) end
    layer.DescendantAdded:Connect(function(descendant)
        if descendant:IsA("ScrollingFrame") then task.defer(consider, descendant) end
    end)
end
SyllinseRuntime.watchOuterMobileScrollFrames(Lumen._Instance)
SyllinseRuntime.watchOuterMobileScrollFrames(Lumen._Overlay)

-- Make each slider's displayed number directly editable without replacing its
-- native Lumen slider. The field is positioned over the existing value readout,
-- and routes numeric edits through the same native Set/Callback path as dragging.
SyllinseRuntime.addNumericSlider = function(section, options)
    local sliderOptions = table.clone(options or {})
    local userCallback = sliderOptions.Callback
    local currentValue = tonumber(sliderOptions.Value) or tonumber(sliderOptions.Min) or 0
    local minimum = tonumber(sliderOptions.Min) or 0
    local maximum = tonumber(sliderOptions.Max) or 100
    local increment = math.max(tonumber(sliderOptions.Increment) or 1, 0.000001)
    local suffix = type(sliderOptions.Suffix) == "string" and string.gsub(sliderOptions.Suffix, "^%s+", "") or ""
    local numericEditor
    local nativeValueLabel

    local function snapValue(value)
        value = tonumber(value)
        if not value then return currentValue end
        value = math.clamp(value, minimum, maximum)
        local steps = math.floor(((value - minimum) / increment) + 0.5)
        value = minimum + steps * increment
        return math.clamp(value, minimum, maximum)
    end

    local function formatNumber(value)
        local formatted = string.format("%.6f", tonumber(value) or currentValue)
        formatted = string.gsub(formatted, "0+$", "")
        formatted = string.gsub(formatted, "%.$", "")
        if formatted == "" or formatted == "-0" then formatted = "0" end
        return formatted
    end

    local function syncEditor(value)
        currentValue = tonumber(value) or currentValue
        if numericEditor and numericEditor.Parent and not numericEditor:IsFocused() then
            numericEditor.Text = formatNumber(currentValue)
        end
    end

    sliderOptions.Callback = function(value)
        currentValue = tonumber(value) or currentValue
        syncEditor(currentValue)
        if type(userCallback) == "function" then
            userCallback(value)
        end
    end

    local control = section:Slider(sliderOptions)
    local nativeSet = control and control.Set

    if type(nativeSet) == "function" then
        control.Set = function(value, ...)
            -- Most Lumen calls use dot-style Set(value); accept colon-style calls too.
            local extraArgs = table.pack(...)
            if value == control then
                value = extraArgs[1]
                local shifted = table.create(math.max(extraArgs.n - 1, 0))
                for i = 2, extraArgs.n do shifted[i - 1] = extraArgs[i] end
                extraArgs = table.pack(table.unpack(shifted))
            end
            local snapped = snapValue(value)
            local ok, result = pcall(nativeSet, snapped, table.unpack(extraArgs, 1, extraArgs.n))
            if not ok then
                -- Compatibility fallback for library revisions that expect self.
                ok, result = pcall(nativeSet, control, snapped, table.unpack(extraArgs, 1, extraArgs.n))
            end
            currentValue = tonumber(control.Value) or snapped
            syncEditor(currentValue)
            if control.Flag and Lumen and Lumen.Flags and Lumen.Flags[control.Flag] then
                Lumen.Flags[control.Flag].Set = control.Set
            end
            return ok and result or nil
        end
        if control.Flag and Lumen and Lumen.Flags and Lumen.Flags[control.Flag] then
            Lumen.Flags[control.Flag].Set = control.Set
        end
    end

    local function parseNumericPrefix(text)
        if type(text) ~= "string" then return nil end
        local valueText = string.match(text, "^%s*([+-]?%d+%.?%d*)")
        return valueText and tonumber(valueText) or nil
    end

    local function findValueLabel(frame)
        if not frame then return nil end
        local objects = {}
        if frame:IsA("TextLabel") or frame:IsA("TextButton") then
            table.insert(objects, frame)
        end
        for _, object in ipairs(frame:GetDescendants()) do
            if object:IsA("TextLabel") or object:IsA("TextButton") then
                table.insert(objects, object)
            end
        end
        local best, bestScore = nil, -math.huge
        for _, object in ipairs(objects) do
            local name = string.lower(object.Name or "")
            local number = parseNumericPrefix(object.Text)
            local score = -math.huge
            if number ~= nil then
                score = 10 - math.min(math.abs(number - currentValue), 1000) / 1000
                if math.abs(number - currentValue) < 0.0001 then score = score + 100 end
            end
            if name == "value" or name == "slidervalue" or name == "valuelabel" then
                score = math.max(score, 500)
            elseif string.find(name, "value", 1, true) or string.find(name, "number", 1, true) then
                score = math.max(score, 300)
            end
            if string.find(name, "title", 1, true) or name == "name" or name == "label" then
                score = score - 500
            end
            if score > bestScore then
                best, bestScore = object, score
            end
        end
        return bestScore > 0 and best or nil
    end

    -- Locate the actual Lumen SliderFrame from the owning section. Lumen's slider
    -- handle does not consistently expose Frame/Instance/Container, so relying on
    -- those fields can silently skip creation of the editor entirely.
    local function findSliderFrame()
        local sectionFrame = section and (section.Frame or section.Instance or section.Container)
        if not (sectionFrame and typeof(sectionFrame) == "Instance" and sectionFrame:IsA("GuiObject")) then
            return nil
        end

        local expectedTitle = string.lower(string.gsub(tostring(sliderOptions.Name or ""), "^%s*(.-)%s*$", "%1"))
        local candidates = {}
        local partialMatch = nil
        for _, object in ipairs(sectionFrame:GetDescendants()) do
            if object:IsA("GuiObject") and string.lower(object.Name or "") == "sliderframe" then
                table.insert(candidates, object)
                local exactTitleMatch = false
                local hasPartialTitleMatch = false
                for _, child in ipairs(object:GetDescendants()) do
                    if child:IsA("TextLabel") or child:IsA("TextButton") then
                        local text = string.lower(string.gsub(tostring(child.Text or ""), "^%s*(.-)%s*$", "%1"))
                        if text == expectedTitle and expectedTitle ~= "" then
                            exactTitleMatch = true
                            break
                        elseif expectedTitle ~= "" and string.find(text, expectedTitle, 1, true) then
                            hasPartialTitleMatch = true
                        end
                    end
                end
                if exactTitleMatch then return object end
                if hasPartialTitleMatch and not partialMatch then partialMatch = object end
            end
        end
        if partialMatch then return partialMatch end
        if #candidates == 1 then return candidates[1] end
        return nil
    end

    task.defer(function()
        local frame
        -- Lumen may finish parenting/layout on the next task step. Re-resolve the
        -- named slider row briefly so every field attaches to its own slider.
        for attempt = 1, 12 do
            frame = findSliderFrame()
            if not (frame and frame:IsA("GuiObject")) then
                local fallback = control and (control.SliderFrame or control.Frame or control.Instance or control.Container)
                if fallback and typeof(fallback) == "Instance" and fallback:IsA("GuiObject") and string.lower(fallback.Name or "") == "sliderframe" then
                    frame = fallback
                end
            end
            if frame and frame.Parent then
                nativeValueLabel = findValueLabel(frame)
                if nativeValueLabel or attempt >= 3 then break end
            end
            task.wait(0.05)
        end

        if not (frame and typeof(frame) == "Instance" and frame:IsA("GuiObject") and frame.Parent) then
            warn("[Syllinse] Could not find the Lumen slider row for numeric editor: " .. tostring(sliderOptions.Name))
            return
        end

        local fieldParent = nativeValueLabel and nativeValueLabel.Parent or frame
        local rightScale, rightOffset = 1, -3
        local topScale, topOffset = 0, 0
        if nativeValueLabel and nativeValueLabel.Parent then
            local labelPosition = nativeValueLabel.Position
            local labelSize = nativeValueLabel.Size
            local anchor = nativeValueLabel.AnchorPoint
            rightScale = labelPosition.X.Scale + (1 - anchor.X) * labelSize.X.Scale
            rightOffset = labelPosition.X.Offset + (1 - anchor.X) * labelSize.X.Offset
            topScale = labelPosition.Y.Scale - anchor.Y * labelSize.Y.Scale
            topOffset = labelPosition.Y.Offset - anchor.Y * labelSize.Y.Offset
            nativeValueLabel.Visible = false
        end

        -- Keep the numeric editor visually indistinguishable from Lumen's own value
        -- readout: compact, transparent, and using the same native type/color. A subtle
        -- fill/stroke appears only while editing so it remains discoverable without
        -- creating a large pill that clashes with the rest of the UI.
        local suffixText = suffix
        local suffixWidth = suffixText == "" and 0 or math.clamp(#suffixText * 5 + 3, 10, 42)
        local digitCount = #formatNumber(currentValue)
        local editorWidth = math.clamp(digitCount * 6 + 5, 24, 42)
        local nativeTextSize = nativeValueLabel and nativeValueLabel.TextSize or 12
        local nativeAbsHeight = nativeValueLabel and nativeValueLabel.AbsoluteSize.Y or 0
        local fieldHeight = math.clamp(nativeAbsHeight > 0 and nativeAbsHeight or (nativeTextSize + 2), 13, 17)
        local valueRightOffset = rightOffset - suffixWidth - (suffixWidth > 0 and 2 or 0)

        local box = Instance.new("TextBox")
        box.Name = "SyllinseNumericSliderInput"
        box.AnchorPoint = Vector2.new(1, 0)
        box.Position = UDim2.new(rightScale, valueRightOffset, topScale, topOffset - 1)
        box.Size = UDim2.fromOffset(editorWidth, fieldHeight)
        box.BackgroundColor3 = Theme.SurfaceHover
        box.BackgroundTransparency = 1
        box.BorderSizePixel = 0
        box.ClearTextOnFocus = false
        box.Text = formatNumber(currentValue)
        box.TextColor3 = nativeValueLabel and nativeValueLabel.TextColor3 or Theme.PanelValue
        box.TextSize = nativeTextSize
        box.TextTransparency = nativeValueLabel and nativeValueLabel.TextTransparency or 0
        box.FontFace = nativeValueLabel and nativeValueLabel.FontFace or FONT_REGULAR
        box.TextXAlignment = Enum.TextXAlignment.Right
        box.TextYAlignment = nativeValueLabel and nativeValueLabel.TextYAlignment or Enum.TextYAlignment.Center
        box.TextWrapped = false
        box.ZIndex = (nativeValueLabel and nativeValueLabel.ZIndex or frame.ZIndex) + 3
        box.Parent = fieldParent
        local editorStroke = Instance.new("UIStroke")
        editorStroke.Name = "SyllinseNumericSliderFocusStroke"
        editorStroke.Color = Theme.Stroke
        editorStroke.Transparency = 1
        editorStroke.Thickness = 1
        editorStroke.Parent = box
        addCorner(box, 3)
        numericEditor = box
        box:SetAttribute("SyllinseNumericSliderInput", true)

        if suffixText ~= "" then
            local suffixLabel = Instance.new("TextLabel")
            suffixLabel.Name = "SyllinseNumericSliderSuffix"
            suffixLabel.AnchorPoint = Vector2.new(1, 0)
            suffixLabel.Position = UDim2.new(rightScale, rightOffset, topScale, topOffset - 1)
            suffixLabel.Size = UDim2.fromOffset(suffixWidth, fieldHeight)
            suffixLabel.BackgroundTransparency = 1
            suffixLabel.BorderSizePixel = 0
            suffixLabel.Text = suffixText
            suffixLabel.TextColor3 = nativeValueLabel and nativeValueLabel.TextColor3 or Theme.PanelSecondary
            suffixLabel.TextSize = nativeTextSize
            suffixLabel.TextTransparency = nativeValueLabel and nativeValueLabel.TextTransparency or 0
            suffixLabel.FontFace = nativeValueLabel and nativeValueLabel.FontFace or FONT_REGULAR
            suffixLabel.TextXAlignment = Enum.TextXAlignment.Left
            suffixLabel.TextYAlignment = nativeValueLabel and nativeValueLabel.TextYAlignment or Enum.TextYAlignment.Center
            suffixLabel.ZIndex = box.ZIndex
            suffixLabel.Parent = fieldParent
        end

        box.Focused:Connect(function()
            box.BackgroundTransparency = 0.35
            editorStroke.Transparency = 0.55
            task.defer(function()
                if box.Parent and box:IsFocused() and #box.Text > 0 then
                    box.SelectionStart = 1
                    box.CursorPosition = #box.Text + 1
                end
            end)
        end)

        box.FocusLost:Connect(function()
            local typed = tonumber(string.match(box.Text or "", "^%s*([+-]?%d+%.?%d*)"))
            if typed ~= nil then
                local snapped = snapValue(typed)
                if control and type(control.Set) == "function" then
                    local ok = pcall(control.Set, snapped)
                    if not ok then pcall(control.Set, control, snapped) end
                end
                currentValue = tonumber(control and control.Value) or snapped
            end
            box.BackgroundTransparency = 1
            editorStroke.Transparency = 1
            syncEditor(currentValue)
        end)

        if nativeValueLabel then
            nativeValueLabel:GetPropertyChangedSignal("Text"):Connect(function()
                local parsed = parseNumericPrefix(nativeValueLabel.Text)
                if parsed ~= nil then syncEditor(parsed) end
            end)
        end
        syncEditor(currentValue)
    end)

    return control
end

SyllinseRuntime.setAutoloadStatusText = function(text)
    local hosts = lumenHosts.Configs
    if not hosts then return false end
    for _, column in pairs(hosts) do
        if typeof(column) == "Instance" then
            for _, descendant in ipairs(column:GetDescendants()) do
                if descendant:IsA("TextLabel")
                    and string.find(descendant.Text, "^Autoload config:") then
                    descendant.Text = text
                    return true
                end
            end
        end
    end
    return false
end

SyllinseRuntime.AUTOLOAD_FILE = "Syllinse_Autoload.txt"
getAutoloadName = function()
    if type(isfile) ~= "function" or type(readfile) ~= "function" then return nil end
    local okExists, exists = pcall(isfile, SyllinseRuntime.AUTOLOAD_FILE)
    if not okExists or not exists then return nil end
    local okRead, value = pcall(readfile, SyllinseRuntime.AUTOLOAD_FILE)
    if okRead and type(value) == "string" then
        value = value:match("^%s*(.-)%s*$")
        if value ~= "" then return value end
    end
    return nil
end

SyllinseRuntime.applyPanelTypography = function(color)
    -- Use Syllinse's original neutral hierarchy: regular controls are muted gray,
    -- section headings slightly brighter, and helper text quieter. Keep the font sizes
    -- readable, but cap dropdown labels so they do not run into the selection field below.
    local controlText = color or Theme.PanelText
    local sectionTitle = Theme.PanelTitle
    local secondaryText = Theme.PanelSecondary
    local iconColor = Theme.PanelIcon
    local valueColor = Theme.PanelValue
    local touchFirst = touchInputEnabled and not UserInputService.KeyboardEnabled

    local function hasDropdownAncestor(object, sectionFrame)
        local current = object.Parent
        for _ = 1, 5 do
            if not current or current == sectionFrame then break end
            local ancestorName = string.lower(current.Name or "")
            if ancestorName:find("dropdown", 1, true)
                or ancestorName:find("drop-down", 1, true)
                or ancestorName == "inputframe"
                or ancestorName == "buttonframe" then
                return true
            end
            current = current.Parent
        end
        return false
    end

    local function makeMobileRowTappable(control, sectionFrame, physicalHeight)
        if not (touchInputEnabled and not UserInputService.KeyboardEnabled) or not control then return end
        Lumen._SyllinseExpandMobileControlRow(control, physicalHeight or 48)
    end

    for _, section in ipairs(Lumen.Sections or {}) do
        local frame = section.Frame
        if frame and frame.Parent then
            for _, descendant in ipairs(frame:GetDescendants()) do
                local name = string.lower(descendant.Name or "")
                local inDropdown = hasDropdownAncestor(descendant, frame)

                if descendant:IsA("TextLabel") or descendant:IsA("TextBox") then
                    local targetColor = controlText
                    local isSectionTitle = descendant.Name == "Title"
                        and descendant.Parent
                        and descendant.Parent.Name == "Header"
                    local isSecondary = name:find("body", 1, true)
                        or name:find("description", 1, true)
                        or name:find("sublabel", 1, true)
                        or name:find("placeholder", 1, true)
                    local isValue = name == "amount" or name:find("value", 1, true)

                    if isSectionTitle then
                        targetColor = sectionTitle
                        descendant.FontFace = FONT_BOLD
                        descendant.TextSize = math.clamp(descendant.TextSize, touchFirst and 18 or 15, touchFirst and 19 or 16)
                        -- Shift only the section title text down by 1 px without moving the icon or divider.
                        -- PaddingTop=2 moves the centered text content down by 1 px and is reused on refresh.
                        local titleYOffset = descendant:FindFirstChild("SyllinseTitleYOffset")
                        if not titleYOffset then
                            titleYOffset = Instance.new("UIPadding")
                            titleYOffset.Name = "SyllinseTitleYOffset"
                            titleYOffset.Parent = descendant
                        end
                        titleYOffset.PaddingTop = UDim.new(0, 2)
                        titleYOffset.PaddingBottom = UDim.new(0, 0)
                        titleYOffset.PaddingLeft = UDim.new(0, 0)
                        titleYOffset.PaddingRight = UDim.new(0, 0)
                    elseif isSecondary then
                        targetColor = secondaryText
                        descendant.FontFace = FONT_REGULAR
                        descendant.TextSize = math.clamp(descendant.TextSize, touchFirst and 15 or 13, touchFirst and 16 or 14)
                    elseif isValue then
                        targetColor = valueColor
                        descendant.FontFace = FONT_REGULAR
                        descendant.TextSize = math.clamp(descendant.TextSize, touchFirst and 16 or 14, touchFirst and 17 or 15)
                    else
                        descendant.FontFace = FONT_REGULAR
                        descendant.TextSize = math.clamp(descendant.TextSize, touchFirst and 16 or 14, touchFirst and 17 or 15)
                    end

                    -- Label text above dropdown fields is the cramped case. A 14 px cap
                    -- keeps those labels inside their original row without moving any UI.
                    if inDropdown and not isValue then
                        descendant.TextSize = math.min(descendant.TextSize, touchFirst and 16 or 14)
                    end

                    if inDropdown and (name == "input" or isValue) then
                        targetColor = Color3.fromRGB(224, 223, 223)
                    end
                    descendant.TextColor3 = targetColor
                    if descendant:IsA("TextBox") then
                        descendant.PlaceholderColor3 = secondaryText
                        descendant.BackgroundColor3 = Theme.SurfaceHover
                    end
                elseif descendant:IsA("TextButton") then
                    descendant.FontFace = FONT_REGULAR
                    -- Color swatches sit inside RightContent. Increase their touch area on
                    -- phones and raise the row height so the following control stays clear.
                    if (touchInputEnabled and not UserInputService.KeyboardEnabled) and descendant.Name == "Button"
                        and descendant.Parent and descendant.Parent.Name == "RightContent" then
                        descendant.Size = UDim2.fromOffset(36 / math.max(responsiveScaleValue, 0.25), 36 / math.max(responsiveScaleValue, 0.25))
                        Lumen._SyllinseExpandMobileControlRow(descendant, 48)
                    end
                    -- Keep the main toggle target comfortably tappable on mobile.
                    if descendant.Name == "Toggle" then
                        makeMobileRowTappable(descendant, frame, 48)
                    elseif descendant.Name == "KeyButton" then
                        makeMobileRowTappable(descendant, frame, 48)
                    elseif descendant.Name == "Click" then
                        makeMobileRowTappable(descendant, frame, 44)
                    end
                    if descendant.Name ~= "Click" then
                        if inDropdown and descendant.Name == "Input" then
                            -- Selected values such as Head and Closest are brighter than helper labels.
                            descendant.TextColor3 = Color3.fromRGB(224, 223, 223)
                            descendant.TextTransparency = 0
                        else
                            descendant.TextColor3 = controlText
                        end
                        descendant.TextSize = math.clamp(descendant.TextSize, touchFirst and 16 or 14, touchFirst and 17 or 15)
                        -- Keep dropdown text legible but within its original field.
                        if inDropdown then
                            descendant.TextSize = math.min(descendant.TextSize, touchFirst and 17 or 15)
                        end
                    else
                        descendant.TextSize = touchFirst and 15 or 13
                        descendant.BackgroundColor3 = Theme.Surface
                        descendant.BackgroundTransparency = 0
                        descendant.AutoButtonColor = false
                        descendant.BorderSizePixel = 0
                        local buttonCorner = descendant:FindFirstChildOfClass("UICorner")
                        if not buttonCorner then
                            buttonCorner = Instance.new("UICorner")
                            buttonCorner.Parent = descendant
                        end
                        buttonCorner.CornerRadius = UDim.new(0, 7)
                        local buttonStroke = descendant:FindFirstChildOfClass("UIStroke")
                        if not buttonStroke then
                            buttonStroke = addStroke(descendant, Theme.Stroke, 0.08, 1)
                        end
                        buttonStroke.Color = Theme.Stroke
                        buttonStroke.Transparency = 0.08
                        if descendant:GetAttribute("SyllinseButtonHoverBound") ~= true then
                            descendant:SetAttribute("SyllinseButtonHoverBound", true)
                            descendant.MouseEnter:Connect(function()
                                if descendant.Parent then
                                    tween(descendant, 0.12, { BackgroundColor3 = Theme.SurfaceHover })
                                end
                            end)
                            descendant.MouseLeave:Connect(function()
                                if descendant.Parent then
                                    tween(descendant, 0.12, { BackgroundColor3 = Theme.Surface })
                                end
                            end)
                        end
                    end
                elseif descendant:IsA("ImageLabel") or descendant:IsA("ImageButton") then
                    -- Lumen's keybind Settings gear can be placed in a non-square control box.
                    -- Fit the image inside that box so its artwork stays proportional on PC
                    -- and mobile, without changing the button's original size or hitbox.
                    if descendant.Name == "Settings"
                        and descendant.Parent and descendant.Parent.Name == "RightContent" then
                        descendant.ScaleType = Enum.ScaleType.Fit
                    end

                    -- Only touch-first devices get larger hit targets. Keep the actual
                    -- gear/bell art proportional by drawing it in a separate fitted label;
                    -- resizing the clickable ImageButton itself must not stretch its image.
                    if (touchInputEnabled and not UserInputService.KeyboardEnabled)
                        and descendant.Parent and descendant.Parent.Name == "RightContent"
                        and (descendant.Name == "Settings"
                            or string.find(descendant.Name, "FeatureNotificationBell_", 1, true) == 1) then
                        local touchArt = descendant:FindFirstChild("SyllinseMobileTouchArt")
                        if not touchArt then
                            local originalSize = descendant.Size
                            local originalWidth = originalSize.X.Scale ~= 0 and descendant.AbsoluteSize.X or originalSize.X.Offset
                            local originalHeight = originalSize.Y.Scale ~= 0 and descendant.AbsoluteSize.Y or originalSize.Y.Offset
                            if originalWidth <= 0 then originalWidth = descendant.AbsoluteSize.X end
                            if originalHeight <= 0 then originalHeight = descendant.AbsoluteSize.Y end
                            originalWidth = math.max(1, originalWidth)
                            originalHeight = math.max(1, originalHeight)

                            local aspect = originalWidth / originalHeight
                            local visualMax = math.floor(math.clamp(math.max(originalWidth, originalHeight) * 1.25, 18, 22) + 0.5)
                            local visualWidth, visualHeight
                            if aspect >= 1 then
                                visualWidth = visualMax
                                visualHeight = visualMax / aspect
                            else
                                visualHeight = visualMax
                                visualWidth = visualMax * aspect
                            end

                            touchArt = Instance.new("ImageLabel")
                            touchArt.Name = "SyllinseMobileTouchArt"
                            touchArt.AnchorPoint = Vector2.new(0.5, 0.5)
                            touchArt.Position = UDim2.fromScale(0.5, 0.5)
                            touchArt.Size = UDim2.fromOffset(math.floor(visualWidth + 0.5), math.floor(visualHeight + 0.5))
                            touchArt.BackgroundTransparency = 1
                            touchArt.BorderSizePixel = 0
                            touchArt.Image = descendant.Image
                            touchArt.ImageColor3 = descendant.ImageColor3
                            touchArt.ImageTransparency = descendant.ImageTransparency
                            touchArt.ScaleType = Enum.ScaleType.Fit
                            touchArt.ZIndex = descendant.ZIndex + 1
                            touchArt.Active = false
                            touchArt.Selectable = false
                            touchArt:SetAttribute("SyllinseOriginalImageTransparency", descendant.ImageTransparency)
                            touchArt.Parent = descendant
                            descendant.ImageTransparency = 1
                        end
                        descendant.Size = UDim2.fromOffset(30, 30)
                        Lumen._SyllinseExpandMobileControlRow(descendant, 40)
                    end

                    if descendant.Name == "SyllinseMobileTouchArt" and descendant.Parent then
                        -- Mirror the parent button's theme tint but preserve the art's
                        -- original transparency and its fitted aspect ratio.
                        descendant.ImageColor3 = descendant.Parent.ImageColor3
                        descendant.ImageTransparency = descendant:GetAttribute("SyllinseOriginalImageTransparency") or 0
                    elseif string.find(descendant.Name, "FeatureNotificationBell_", 1, true) == 1 then
                        local featureFlag = descendant:GetAttribute("FeatureFlag")
                        local enabled = featureFlag and SyllinseRuntime.featureNotificationSettings[featureFlag] == true
                        descendant.ImageColor3 = enabled and Theme.Text or Theme.Dim
                    elseif name == "sliderbar" then
                        descendant.ImageColor3 = Theme.SliderTrack
                    else
                        descendant.ImageColor3 = iconColor
                    end
                    local touchArt = descendant:FindFirstChild("SyllinseMobileTouchArt")
                    if touchArt then
                        touchArt.ImageColor3 = descendant.ImageColor3
                        touchArt.ImageTransparency = touchArt:GetAttribute("SyllinseOriginalImageTransparency") or 0
                    end
                elseif descendant:IsA("Frame") then
                    if name == "button" and descendant.Parent and descendant.Parent.Name == "SliderFrame" then
                        descendant.BackgroundColor3 = Theme.SliderTrack
                    elseif name == "circle" then
                        descendant.BackgroundColor3 = Theme.SliderHandle
                    end
                end
            end
        end
    end
end

-- Keep Lumen popups and overlay text consistent with the Syllinse shell.
-- Use FontFace + explicit weights so letterforms are consistent across all UI layers.
SyllinseRuntime.applyLumenTypeface = function(instance)
    if not instance then return end
    local function styleText(object)
        if not (object:IsA("TextLabel") or object:IsA("TextButton") or object:IsA("TextBox")) then
            return
        end
        local name = string.lower(object.Name or "")
        local parentName = object.Parent and string.lower(object.Parent.Name or "") or ""
        local isHeader = (name == "title" and parentName == "header")
            or name == "toasttitle"
            or name == "notificationtitle"
            or name == "paragraphheader"
            or name == "headertext"
        object.FontFace = isHeader and FONT_BOLD or FONT_REGULAR
        object.TextScaled = false
        object.AutoLocalize = false
    end
    styleText(instance)
    for _, descendant in ipairs(instance:GetDescendants()) do
        styleText(descendant)
    end
end

SyllinseRuntime.applyAccentColor = function(color)
    if typeof(color) ~= "Color3" then return end
    Theme.Accent = color
    Theme.IconActive = color
    -- Match the launcher logo's restrained backlight to the active UI accent.
    -- Keep the original low-intensity glow; only its hue follows the theme.
    if launcherGlow and launcherGlow.Parent then launcherGlow.ImageColor3 = color end
    if launcherGlowCyan and launcherGlowCyan.Parent then launcherGlowCyan.ImageColor3 = color end
    if launcherGlowViolet and launcherGlowViolet.Parent then launcherGlowViolet.ImageColor3 = color end
    Lumen.Theme.Accent = color
    Lumen.ApplyTheme()
    SyllinseRuntime.applyPanelTypography(Theme.PanelText)
    for _, nav in ipairs(navButtons) do
        local selected = nav.Name == activeName
        if nav.Icon then nav.Icon.ImageColor3 = selected and color or Theme.IconIdle end
        local label = terminologyLabels[nav.Name]
        if label then
            local labelColor = selected and terminologyTint(Theme.IconActive, 0.76)
                or (nav.Hovering and terminologyTint(Theme.IconHover, 0.86) or Theme.IconIdle)
            tween(label, 0.14, { TextColor3 = labelColor })
        end
        setFallbackEmphasis(nav.Button, selected)
        if nav.Gear then setGearIconColor(nav.Gear, selected and color or Theme.IconIdle) end
    end
    for tabName, entry in pairs(headerIcons) do
        local selected = tabName == activeName
        -- Keep the title-row icon matched to its neutral gray title, even if the user
        -- changes the separate sidebar accent color in Settings.
        if entry.Icon then entry.Icon.ImageColor3 = selected and Theme.HeaderIconTint or Theme.PanelIcon end
        recolorFallback(
            entry.Holder:FindFirstChild("ImageFallback"),
            selected,
            Theme.PanelTitle,
            Theme.PanelIcon
        )
    end
    if headerGear then setGearIconColor(headerGear, color) end
end

-- Live feature search across Lumen's real searchable control registry. Category prefixes
-- navigate to the matching Syllinse tab; all other terms filter Lumen rows in place.
SyllinseRuntime.startsWith = function(text, prefix)
    return text:sub(1, #prefix) == prefix
end
SyllinseRuntime.filterLumenControls = function(query)
    query = string.lower(query or "")
    local sectionMatches = {}
    local pageMatches = {}

    if query == "" then
        for _, item in ipairs(Lumen.Searchable or {}) do
            if item.Frame and item.Frame.Parent then item.Frame.Visible = true end
        end
        for _, section in ipairs(Lumen.Sections or {}) do
            if section.Frame and section.Frame.Parent then section.Frame.Visible = true end
        end
        return
    end

    for _, section in ipairs(Lumen.Sections or {}) do
        local matched = string.find(string.lower(tostring(section.Name or "")), query, 1, true) ~= nil
        sectionMatches[section] = matched
        if section.Frame and section.Frame.Parent then section.Frame.Visible = matched end
        if matched and section.PageFrame then
            local tabName = section.PageFrame:GetAttribute("SyllinseTabName")
            if tabName then pageMatches[tabName] = true end
        end
    end

    for _, item in ipairs(Lumen.Searchable or {}) do
        local frame = item.Frame
        if frame and frame.Parent then
            local section = item.Section
            local text = string.lower(tostring(item.Text or ""))
            local match = string.find(text, query, 1, true) ~= nil or (section and sectionMatches[section] == true)
            frame.Visible = match
            if match and section and section.Frame and section.Frame.Parent then
                section.Frame.Visible = true
            end
            if match and section and section.PageFrame then
                local tabName = section.PageFrame:GetAttribute("SyllinseTabName")
                if tabName then pageMatches[tabName] = true end
            end
        end
    end

    local currentHasMatch = pageMatches[activeName] == true
    if not currentHasMatch then
        for _, item in ipairs(navItems) do
            if pageMatches[item.Name] then
                setActiveTab(item.Name)
                break
            end
        end
    end
end

searchBox:GetPropertyChangedSignal("Text"):Connect(function()
    local query = string.lower((searchBox.Text or ""):match("^%s*(.-)%s*$") or "")
    if query == "" then
        currentSearchMode = "none"
        SyllinseRuntime.filterLumenControls("")
        return
    end

    local tabPrefix
    for _, item in ipairs(navItems) do
        local name = string.lower(item.Name)
        if SyllinseRuntime.startsWith(name, query) then
            tabPrefix = item.Name
            break
        end
    end
    if tabPrefix then
        currentSearchMode = "tab"
        SyllinseRuntime.filterLumenControls("")
        if activeName ~= tabPrefix then setActiveTab(tabPrefix) end
    else
        currentSearchMode = "controls"
        SyllinseRuntime.filterLumenControls(query)
    end
end)

    if SyllinseRuntime.uiToggleKeyListening then

        return
    end
    if gameProcessed then return end
